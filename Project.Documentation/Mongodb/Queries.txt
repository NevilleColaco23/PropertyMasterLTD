# To delete documents with _id greater than 3
db.Users.deleteMany({

});


# Add a new field to all documents (set a constant value)
js
db.yourCollection.updateMany(
  {},                       // match all documents
  { $set: { emailConfirmed: false } }
)


# Add a field only if it doesn’t already exist
js
db.yourCollection.updateMany(
  { emailConfirmed: { $exists: false } },
  { $set: { emailConfirmed: false } }
)
